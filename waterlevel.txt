// =========================================================================================
// LIGHTWEIGHT JSON PARSER HELPER
// =========================================================================================

String parseJsonString(String json, String key) {
    String searchKey = "\"" + key + "\":\"";
    int startIdx = json.indexOf(searchKey);
    if (startIdx == -1) {
        searchKey = "\"" + key + "\":";
        startIdx = json.indexOf(searchKey);
        if (startIdx == -1) return "";
        startIdx += searchKey.length();
        while (startIdx < json.length() && (json[startIdx] == ' ' || json[startIdx] == '\t')) {
            startIdx++;
        }
        int endIdx = startIdx;
        while (endIdx < json.length() && (isDigit(json[endIdx]) || json[endIdx] == '.' || json[endIdx] == '-')) {
            endIdx++;
        }
        return json.substring(startIdx, endIdx);
    }
    startIdx += searchKey.length();
    int endIdx = json.indexOf("\"", startIdx);
    if (endIdx == -1) return "";
    return json.substring(startIdx, endIdx);
}

// Extract version string (e.g. "1.701") from binary URL / filename
String extractVersionFromUrl(String url) {
    int lastSlash = url.lastIndexOf('/');
    String filename = (lastSlash != -1) ? url.substring(lastSlash + 1) : url;
    int dotIdx = filename.indexOf('.');
    if (dotIdx != -1) {
        int vStart = dotIdx;
        while (vStart > 0 && (isDigit(filename[vStart - 1]) || filename[vStart - 1] == 'v' || filename[vStart - 1] == 'V')) {
            vStart--;
        }
        int vEnd = dotIdx + 1;
        while (vEnd < filename.length() && isDigit(filename[vEnd])) {
            vEnd++;
        }
        if (vStart < dotIdx && vEnd > dotIdx + 1) {
            String ver = filename.substring(vStart, vEnd);
            if (ver.startsWith("v") || ver.startsWith("V")) {
                ver = ver.substring(1);
            }
            return ver;
        }
    }
    return "";
}

// =========================================================================================

float getDistanceRaw() {
    digitalWrite(pinTrig, LOW);
    delayMicroseconds(4);
    digitalWrite(pinTrig, HIGH);
    delayMicroseconds(10);
    digitalWrite(pinTrig, LOW);
    
    // Echo pulse measurement (30ms timeout ~= 5 meters maximum range)
    long duration = pulseIn(pinEcho, HIGH, 30000);
    if (duration == 0) {
        return -1.0; // Timeout or sensor out of range
    }
    
    // Speed of sound: 0.0343 cm/us divided by 2 (round-trip)
    return (duration * 0.0343) / 2.0;
}

float getFilteredDistance() {
    float raw = getDistanceRaw();
    if (raw < 0.0 || raw > (tankHeight + sensorHeight) + 30.0) {
        return (smoothedDistance >= 0.0) ? smoothedDistance : (tankHeight + sensorHeight);
    }
    
    if (smoothedDistance < 0.0) {
        smoothedDistance = raw;
    } else {
        smoothedDistance = (raw * filterAlpha) + (smoothedDistance * (1.0 - filterAlpha));
    }
    return smoothedDistance;
}

// =========================================================================================
// WI-FI NETWORK ACTIVE SCANNER
// =========================================================================================

void performWifiScan() {
    Serial.println("[SCAN] Initiating active 2.4GHz Wi-Fi scan...");
    
    WiFi.disconnect(false, false);
    delay(50);
    
    int n = WiFi.scanNetworks(false, false, false, 250);
    scannedNetworks.clear();
    
    if (n > 0) {
        for (int i = 0; i < n; ++i) {
            String ssid = WiFi.SSID(i);
            ssid.trim();
            if (ssid.length() == 0 || ssid.startsWith("ESP32_HydroSync")) {
                continue;
            }
            
            bool found = false;
            for (auto &net : scannedNetworks) {
                if (net.ssid == ssid) {
                    if (WiFi.RSSI(i) > net.rssi) {
                        net.rssi = WiFi.RSSI(i);
                        net.encryptionType = WiFi.encryptionType(i);
                    }
                    found = true;
                    break;
                }
            }
            
            if (!found) {
                WifiNetworkInfo info;
                info.ssid = ssid;
                info.rssi = WiFi.RSSI(i);
                info.encryptionType = WiFi.encryptionType(i);
                scannedNetworks.push_back(info);
            }
        }
        
        std::sort(scannedNetworks.begin(), scannedNetworks.end(), [](const WifiNetworkInfo &a, const WifiNetworkInfo &b) {
            return a.rssi > b.rssi;
        });
        
        Serial.printf("[SCAN] Discovered %d unique Wi-Fi access points.\n", (int)scannedNetworks.size());
    } else {
        Serial.println("[SCAN] No Wi-Fi networks found or scan timed out.");
    }
    
    WiFi.scanDelete();
    lastScanTimestamp = millis();
}

String getWifiScanJson() {
    String json = "{\"success\":true,\"networks\":[";
    for (size_t i = 0; i < scannedNetworks.size(); ++i) {
        if (i > 0) json += ",";
        json += "{\"ssid\":\"" + scannedNetworks[i].ssid + "\",";
        json += "\"rssi\":" + String(scannedNetworks[i].rssi) + ",";
        json += "\"secure\":" + String(scannedNetworks[i].encryptionType != WIFI_AUTH_OPEN ? "true" : "false") + "}";
    }
    json += "]}";
    return json;
}

    html.replace("%TANK_DIAMETER%", String(tankDiameter, 1));
    html.replace("%NUM_TANKS%", String(numTanks));
    html.replace("%INTERVAL%", String(telemetryInterval));
    html.replace("%GSM_NUMS%", gsmNumbersList);
    html.replace("%PIN_LED%", String(pinLed));
    html.replace("%PIN_TRIG%", String(pinTrig));
    html.replace("%PIN_ECHO%", String(pinEcho));
    html.replace("%LED_ACT_LOW_SEL%", ledActiveLow ? "selected" : "");
    html.replace("%LED_ACT_HIGH_SEL%", !ledActiveLow ? "selected" : "");

    preferences.begin(PREF_NAMESPACE, true);
    String lastOta = preferences.getString("last_ota_url", "");
    String lastVer = preferences.getString("last_ota_ver", "");
    preferences.end();

    String otaStatusHtml = "";
    if (lastOta.length() > 0) {
        otaStatusHtml = "<span style='color:#06b6d4;font-weight:bold;'>Locked & Applied:</span> " + lastOta;
        if (lastVer.length() > 0) {
            otaStatusHtml += " (v" + lastVer + ")";
        }
        otaStatusHtml += "<br><a href='/clear-ota-flag' onclick=\"return confirm('Clear OTA version flag from device memory?');\" style='color:#f43f5e;font-weight:bold;display:inline-block;margin-top:6px;text-decoration:underline;'>[Clear OTA Flag]</a>";
    } else {
        otaStatusHtml = "<span style='color:#10b981;'>Initial Factory Firmware</span> &bull; No OTA update recorded.";
    }
    html.replace("%LAST_OTA_STATUS%", otaStatusHtml);

    return html;
}

void setupWebServer() {
    server.on("/", HTTP_GET, []() {
        server.send(200, "text/html", renderPortalPage());
    });

    server.on("/clear-ota-flag", HTTP_GET, []() {
        preferences.begin(PREF_NAMESPACE, false);
        preferences.remove("last_ota_url");
        preferences.remove("last_ota_ver");
        preferences.remove("ota_applied");
        preferences.remove("pending_ota");
        preferences.remove("pending_ota_ver");
        preferences.remove("ota_fail_cnt");
        preferences.end();
        Serial.println("[OTA] Device Memory OTA Version Flag cleared by user.");
        server.sendHeader("Location", "/");
        server.send(303);
    });

    server.on("/api/scan", HTTP_GET, []() {
        if (millis() - lastScanTimestamp > 10000 || scannedNetworks.empty()) {
            performWifiScan();
        }
        server.send(200, "application/json", getWifiScanJson());
    });

    server.on("/rescan", HTTP_GET, []() {
        performWifiScan();
        server.sendHeader("Location", "/");

// =========================================================================================
// OVER-THE-AIR (OTA) FIRMWARE FLASHING WITH DEVICE MEMORY FLAG PROTECTION
// =========================================================================================

void triggerHttpOta(String url) {
    url.trim();
    if (url.length() == 0 || (!url.startsWith("http://") && !url.startsWith("https://"))) {
        Serial.println("[OTA] Aborted: Invalid OTA URL format.");
        return;
    }

    // Safety check against device flash memory: prevent redundant re-flashing loop
    preferences.begin(PREF_NAMESPACE, false);
    String lastOtaUrl = preferences.getString("last_ota_url", "");
    String lastOtaVer = preferences.getString("last_ota_ver", "");
    preferences.end();

    if (lastOtaUrl.length() > 0 && url.equalsIgnoreCase(lastOtaUrl)) {
        Serial.printf("[OTA] Skipped: Firmware URL '%s' is already marked as applied in device memory.\n", url.c_str());
        return;
    }

    Serial.printf("[OTA] Commencing Over-The-Air firmware upgrade from: %s\n", url.c_str());
    setLedHardware(false);

    // Save pending_ota flag to flash memory before flashing begins
    preferences.begin(PREF_NAMESPACE, false);
    preferences.putString("pending_ota", url);
    String targetVer = extractVersionFromUrl(url);
    if (targetVer.length() > 0) {
        preferences.putString("pending_ota_ver", targetVer);
    }
    preferences.end();

    WiFiClientSecure clientSecure;
    clientSecure.setInsecure();
    
    // Explicitly set rebootOnUpdate to false so we can commit the version flag to NVS before restarting
    httpUpdate.rebootOnUpdate(false);
    t_httpUpdate_return result = httpUpdate.update(clientSecure, url);

    switch (result) {
        case HTTP_UPDATE_FAILED:
            Serial.printf("[OTA] Update Failed! Error (%d): %s\n", 
                          httpUpdate.getLastError(), httpUpdate.getLastErrorString().c_str());
            preferences.begin(PREF_NAMESPACE, false);
            preferences.remove("pending_ota");
            preferences.remove("pending_ota_ver");
            // Count download/flashing failures to protect against rapid bootloop on 404 or corrupt URLs
            {
                int failCnt = preferences.getInt("ota_fail_cnt", 0) + 1;
                preferences.putInt("ota_fail_cnt", failCnt);
                if (failCnt >= 3) {
                    Serial.println("[OTA] Warning: Repeated OTA download failures. Marking URL as skipped in device memory.");
                    preferences.putString("last_ota_url", url);
                    preferences.putInt("ota_fail_cnt", 0);
                }
            }
            preferences.end();
            setLedHardware(WiFi.status() == WL_CONNECTED);
            break;

        case HTTP_UPDATE_NO_UPDATES:
            Serial.println("[OTA] Server reported no updates available.");
            preferences.begin(PREF_NAMESPACE, false);
            preferences.remove("pending_ota");
            preferences.remove("pending_ota_ver");
            preferences.end();
            setLedHardware(WiFi.status() == WL_CONNECTED);
            break;

        case HTTP_UPDATE_OK:
            Serial.println("[OTA] Firmware Flashing Successfully Completed!");
            Serial.printf("[OTA] Storing updated version flag for '%s' in device NVS flash memory...\n", url.c_str());
            preferences.begin(PREF_NAMESPACE, false);
            preferences.putString("last_ota_url", url);
            if (targetVer.length() > 0) {
                preferences.putString("last_ota_ver", targetVer);
            } else {
                preferences.putString("last_ota_ver", FIRMWARE_VERSION);
            }
            preferences.putBool("ota_applied", true);
            preferences.remove("pending_ota");
            preferences.remove("pending_ota_ver");
            preferences.putInt("ota_fail_cnt", 0);
            preferences.end();
            Serial.println("[OTA] Version flag safely committed to device memory. Rebooting controller now...");
            delay(600);
            ESP.restart();
            break;
    }
}

// =========================================================================================
// HTTP TELEMETRY POST (COMMUNICATING DIRECTLY WITH THIS WEB APP)
// =========================================================================================

void handleHttpResponseConfig(String response) {
    if (response.indexOf("\"success\":true") == -1) return;

    preferences.begin(PREF_NAMESPACE, false);
    bool changed = false;

    // 1. Device ID Sync
    String devStr = parseJsonString(response, "device_id");
    if (devStr.length() > 0 && devStr != deviceId) {
        deviceId = devStr;
        preferences.putString("deviceId", devStr);
        Serial.printf("[HTTP Sync] Updated Device ID: %s\n", devStr.c_str());
        changed = true;
    }

    // 1b. WiFi Network Credentials Sync
    String newSsid = parseJsonString(response, "wifi_ssid");
    if (newSsid.length() > 0 && newSsid != wifiSSID) {
        wifiSSID = newSsid;
        preferences.putString("ssid", wifiSSID);
        Serial.printf("[HTTP Sync] Updated WiFi SSID: %s\n", wifiSSID.c_str());
        changed = true;
    }

    String newPass = parseJsonString(response, "wifi_pass");
    if (newPass.length() > 0 && newPass != wifiPASS) {
        wifiPASS = newPass;
        preferences.putString("pass", wifiPASS);
        Serial.printf("[HTTP Sync] Updated WiFi Password in NVS.\n");
        changed = true;
    }

    // 2. Tank Height
    String heightStr = parseJsonString(response, "tank_height");
    if (heightStr.length() > 0) {
        float val = heightStr.toFloat();
        if (val > 0 && val != tankHeight) {
            tankHeight = val;
            preferences.putFloat("height", val);
            Serial.printf("[HTTP Sync] Updated tank height: %.1f cm\n", val);
            changed = true;
        }
    }

    // 3. Sensor Offset Height
    String shStr = parseJsonString(response, "sensor_height");
    if (shStr.length() > 0) {
        float val = shStr.toFloat();
        if (val >= 0 && val != sensorHeight) {
            sensorHeight = val;
            preferences.putFloat("sensor_height", val);
            Serial.printf("[HTTP Sync] Updated sensor offset: %.1f cm\n", val);
            changed = true;
        }
    }

    // 4. Tank Diameter
    String diaStr = parseJsonString(response, "tank_diameter");
    if (diaStr.length() > 0) {
        float val = diaStr.toFloat();
        if (val > 0 && val != tankDiameter) {
            tankDiameter = val;
            preferences.putFloat("diameter", val);
            Serial.printf("[HTTP Sync] Updated tank diameter: %.1f cm\n", val);
            changed = true;
        }
    }

    // 5. Num Tanks
    String tanksStr = parseJsonString(response, "num_tanks");
    if (tanksStr.length() > 0) {
        int val = tanksStr.toInt();
        if (val > 0 && val != numTanks) {
            numTanks = val;
            preferences.putInt("tanks", val);
            Serial.printf("[HTTP Sync] Updated num tanks: %d\n", val);
            changed = true;
        }
    }

    // 6. Telemetry Interval
    String intervalStr = parseJsonString(response, "telemetry_interval");
    if (intervalStr.length() > 0) {
        int val = intervalStr.toInt();
        if (val >= 5 && val != telemetryInterval) {
            telemetryInterval = val;
            preferences.putInt("interval", val);
            Serial.printf("[HTTP Sync] Updated telemetry interval: %d seconds\n", val);
            changed = true;
        }
    }

    // 7. GSM Whitelist
    String gsmStr = parseJsonString(response, "gsm_numbers");
    if (gsmStr.length() > 0 && gsmStr != gsmNumbersList) {
        gsmNumbersList = gsmStr;
        preferences.putString("gsm_nums", gsmStr);
        Serial.printf("[HTTP Sync] Updated GSM whitelist: %s\n", gsmStr.c_str());
        changed = true;
    }

    // 8. Pending OTA URL with Device Memory Guard
    String otaUrl = parseJsonString(response, "ota_url");
    otaUrl.trim();
    if (otaUrl.length() > 0 && (otaUrl.startsWith("http://") || otaUrl.startsWith("https://"))) {
        String lastOtaUrl = preferences.getString("last_ota_url", "");
        String lastOtaVer = preferences.getString("last_ota_ver", "");

        bool alreadyApplied = false;
        if (lastOtaUrl.length() > 0 && otaUrl.equalsIgnoreCase(lastOtaUrl)) {
            alreadyApplied = true;
        } else if (otaUrl.indexOf(FIRMWARE_VERSION) != -1) {
            alreadyApplied = true;
        } else if (lastOtaVer.length() > 0 && otaUrl.indexOf(lastOtaVer) != -1) {
            alreadyApplied = true;
        }

        if (alreadyApplied) {
            Serial.printf("[HTTP Sync] OTA Firmware '%s' already applied on this device (Device Memory Flag Active). Skipping duplicate flash.\n", otaUrl.c_str());
        } else {
            Serial.printf("[HTTP Sync] New Pending OTA Firmware URL detected: %s\n", otaUrl.c_str());
            preferences.end();
            triggerHttpOta(otaUrl);
            return;
        }
    }

    preferences.end();
    if (changed) {
        Serial.println("[HTTP Sync] Configuration changes committed to NVS.");
    }
}

void sendHttpTelemetry(float depth, float volume) {
    if (WiFi.status() != WL_CONNECTED || WiFi.localIP() == IPAddress(0, 0, 0, 0) || apiServerUrl.length() == 0) {
        return;
    }

    // Ensure fallback DNS in case router DHCP provided 0.0.0.0
    if (WiFi.dnsIP() == IPAddress(0, 0, 0, 0)) {
        WiFi.config(INADDR_NONE, INADDR_NONE, INADDR_NONE, IPAddress(8, 8, 8, 8), IPAddress(1, 1, 1, 1));
    }

    HTTPClient http;
    String endpoint = apiServerUrl;
    if (!endpoint.endsWith("/")) {
        endpoint += "/";
    }
    endpoint += "api/telemetry";

    // Read stored OTA flags from NVS to acknowledge applied status to server
    preferences.begin(PREF_NAMESPACE, true);
    String lastOtaUrl = preferences.getString("last_ota_url", "");
    bool otaApplied = preferences.getBool("ota_applied", false);
    preferences.end();

    String jsonPayload = "{\"device_id\":\"" + deviceId + "\","
                         "\"level\":" + String(depth, 1) + ","
                         "\"volume\":" + String(volume, 1) + ","
                         "\"data_usage\":" + String(totalBytesSent) + ","
                         "\"local_ip\":\"" + WiFi.localIP().toString() + "\","
                         "\"version\":\"" + String(FIRMWARE_VERSION) + "\"";
    if (lastOtaUrl.length() > 0) {
        jsonPayload += ",\"last_ota_url\":\"" + lastOtaUrl + "\"";
    }
    if (otaApplied) {
        jsonPayload += ",\"ota_status\":\"applied\"";
    }
    jsonPayload += "}";

    int httpResponseCode = -1;

    Serial.printf("[HTTP API] Transmitting telemetry to: %s\n", endpoint.c_str());

    if (endpoint.startsWith("https://")) {
        WiFiClientSecure clientSecure;
        clientSecure.setInsecure();
        clientSecure.setTimeout(10);
        http.begin(clientSecure, endpoint);
        http.setTimeout(10000);
        http.addHeader("Content-Type", "application/json");
        httpResponseCode = http.POST(jsonPayload);

        if (httpResponseCode > 0) {
            totalBytesSent += (jsonPayload.length() + 220); // Payload bytes + HTTP header overhead
            Serial.printf("[HTTP API] Telemetry sent (HTTP %d).\n", httpResponseCode);
            if (httpResponseCode == 200) {
                String responseBody = http.getString();
                handleHttpResponseConfig(responseBody);
            }
        } else {
            Serial.printf("[HTTP API] Telemetry POST failed (%s) to %s\n", http.errorToString(httpResponseCode).c_str(), endpoint.c_str());
            if (httpResponseCode == -1) {
                Serial.println("[HTTP Note] Connection refused usually means router WAN/internet uplink is still warming up after Wi-Fi reconnect. Will retry automatically.");
                delay(60000);
                pendingReboot = true;
            }
        }
        http.end();
        clientSecure.stop(); // Release TLS socket immediately from LwIP
    } else {
        WiFiClient client;
        client.setTimeout(10);
        http.begin(client, endpoint);
        http.setTimeout(10000);
        http.addHeader("Content-Type", "application/json");
        httpResponseCode = http.POST(jsonPayload);

        if (httpResponseCode > 0) {
            totalBytesSent += (jsonPayload.length() + 220); // Payload bytes + HTTP header overhead
            Serial.printf("[HTTP API] Telemetry sent (HTTP %d).\n", httpResponseCode);
            if (httpResponseCode == 200) {
                String responseBody = http.getString();
                handleHttpResponseConfig(responseBody);
            }
        } else {
            Serial.printf("[HTTP API] Telemetry POST failed (%s) to %s\n", http.errorToString(httpResponseCode).c_str(), endpoint.c_str());
        }
        http.end();
        client.stop(); // Release TCP socket immediately
    }

    // Periodic ping to keep server-side SMS schedules synchronized
    static unsigned long lastCronCallTime = 0;
    unsigned long nowMs = millis();
    if (lastCronCallTime == 0 || nowMs - lastCronCallTime >= 60000) {
        lastCronCallTime = nowMs;
        HTTPClient cronHttp;
        String cronEndpoint = apiServerUrl;
        if (!cronEndpoint.endsWith("/")) cronEndpoint += "/";
        cronEndpoint += "api/sms/cron";

        if (cronEndpoint.startsWith("https://")) {
            WiFiClientSecure cronSec;
            cronSec.setInsecure();
            cronSec.setTimeout(6);
            cronHttp.begin(cronSec, cronEndpoint);
            cronHttp.setTimeout(6000);
            cronHttp.GET();
            cronHttp.end();
            cronSec.stop();
        } else {
            WiFiClient cronCl;
            cronCl.setTimeout(6);
            cronHttp.begin(cronCl, cronEndpoint);
            cronHttp.setTimeout(6000);
            cronHttp.GET();
            cronHttp.end();
            cronCl.stop();
        }
    }
}

void initGSM() {
#if ENABLE_GSM_HARDWARE
    Serial.printf("[GSM] Initializing GSM on RX=%d, TX=%d...\n", GSM_RX_PIN, GSM_TX_PIN);
    SerialGSM.begin(9600, SERIAL_8N1, GSM_RX_PIN, GSM_TX_PIN);
    delay(500);

    SerialGSM.println("AT");
    delay(200);
    SerialGSM.println("ATE0");
    delay(200);
    SerialGSM.println("AT+CMGF=1");
    delay(200);
    SerialGSM.println("AT+CLIP=1");
    delay(200);

    gsmInitialized = true;
    Serial.println("[GSM] GSM interface initialized.");
#else
    // GSM hardware disabled: prevents hijacking of GPIO pins or wasting CPU cycles
#endif
}

bool isNumberAuthorized(String callingNumber) {
    if (callingNumber.length() == 0 || gsmNumbersList.length() == 0) return false;
    callingNumber.trim();
    String whitelist = gsmNumbersList;

    int startIdx = 0;
    while (true) {
        int commaIdx = whitelist.indexOf(',', startIdx);
        String num = (commaIdx == -1) ? whitelist.substring(startIdx) : whitelist.substring(startIdx, commaIdx);
        num.trim();

        if (num.length() > 0) {
            if (callingNumber == num) return true;
            if (num.length() >= 9 && callingNumber.length() >= 9) {
                if (num.substring(num.length() - 9) == callingNumber.substring(callingNumber.length() - 9)) {
                    return true;
                }
            }
        }

        if (commaIdx == -1) break;
        startIdx = commaIdx + 1;
    }
    return false;
}

void sendSmsResponse(String recipient, String message) {
#if ENABLE_GSM_HARDWARE
    recipient.trim();
    if (recipient.length() == 0 || !gsmInitialized) return;

    Serial.printf("[GSM] Sending SMS to %s: %s\n", recipient.c_str(), message.c_str());
    SerialGSM.println("AT+CMGS=\"" + recipient + "\"");
    delay(300);
    SerialGSM.print(message);
    delay(100);
    SerialGSM.write(26);
    delay(1000);
#endif
}

void processGsmLine(String line) {
    line.trim();
    if (line.length() == 0) return;

    if (line.startsWith("+CLIP:")) {
        int firstQuote = line.indexOf('\"');
        int secondQuote = line.indexOf('\"', firstQuote + 1);
        if (firstQuote != -1 && secondQuote != -1) {
            String caller = line.substring(firstQuote + 1, secondQuote);
            Serial.printf("[GSM] Incoming Call from: %s\n", caller.c_str());

            if (isNumberAuthorized(caller)) {
                Serial.println("[GSM] Authorized caller. Sending status SMS...");
                delay(1000);
                SerialGSM.println("ATH");
                delay(500);

                float depth = (currentDepthValue > 0) ? currentDepthValue : 0.0;
                float volume = (currentVolumeValue > 0) ? currentVolumeValue : 0.0;
                float percent = (tankHeight > 0) ? ((depth / tankHeight) * 100.0) : 0.0;
                if (percent > 100.0) percent = 100.0;

                String sms = "HydroSync [" + deviceId + "]:\n";
                sms += "Water Level: " + String(percent, 1) + "%\n";
                sms += "Depth: " + String(depth, 1) + " cm\n";
                sms += "Volume: " + String(volume, 0) + " L\n";
                sms += "Wi-Fi: " + String(WiFi.status() == WL_CONNECTED ? "Connected" : "Disconnected");

                sendSmsResponse(caller, sms);
            } else {
                Serial.println("[GSM] Caller not in whitelist. Dropping call.");
                SerialGSM.println("ATH");
            }
        }
    }
}

void updateGSMModule() {
#if ENABLE_GSM_HARDWARE
    if (!gsmInitialized) return;
    while (SerialGSM.available()) {
        char c = SerialGSM.read();
        if (c == '\n') {
            processGsmLine(gsmBuffer);
            gsmBuffer = "";
        } else if (c != '\r') {
            gsmBuffer += c;
            if (gsmBuffer.length() > 256) gsmBuffer = "";
        }
    }
#endif
}

void setup() {
    // 1. Initialize Serial Console immediately
    Serial.begin(115200);
    Serial.setTxTimeoutMs(0); // Prevent blocking if serial terminal is closed

    // Give USB-CDC terminal time to enumerate on modern computers
    delay(1000);

    // Initial banner
    Serial.println();
    Serial.println();
    Serial.println("==================================================");
    Serial.println("        IoT HydroSync Water Level Controller");
    Serial.printf("            Firmware Version: %s\n", FIRMWARE_VERSION);
    Serial.println("==================================================");
    Serial.flush();

#if defined(CONFIG_IDF_TARGET_ESP32C3) || defined(ESP32C3)
    Serial.println("[ARCH] Detected Board Architecture: ESP32-C3 (RISC-V)");
#else
    Serial.println("[ARCH] Detected Board Architecture: Classic ESP32 (Xtensa)");
#endif

    // 2. Load persistent calibration and pin configurations from NVS
    preferences.begin(PREF_NAMESPACE, false);

    // OTA Version Flag Check: if controller just rebooted from an OTA flash, finalize the flag
    String pendingOta = preferences.getString("pending_ota", "");
    String pendingVer = preferences.getString("pending_ota_ver", "");
    if (pendingOta.length() > 0) {
        Serial.printf("[OTA] Fresh boot verified following OTA update: %s\n", pendingOta.c_str());
        preferences.putString("last_ota_url", pendingOta);
        if (pendingVer.length() > 0) {
            preferences.putString("last_ota_ver", pendingVer);
        } else {
            preferences.putString("last_ota_ver", FIRMWARE_VERSION);
        }
        preferences.putBool("ota_applied", true);
        preferences.remove("pending_ota");
        preferences.remove("pending_ota_ver");
        preferences.putInt("ota_fail_cnt", 0);
    }
    String lastOtaUrl = preferences.getString("last_ota_url", "");
    String lastOtaVer = preferences.getString("last_ota_ver", "");

    wifiSSID = preferences.getString("ssid", "");
    wifiPASS = preferences.getString("pass", "");
    deviceId = preferences.getString("deviceId", "plg_paribogika");
    apiServerUrl = preferences.getString("api_url", "");
    tankHeight = preferences.getFloat("height", 200.0);
    sensorHeight = preferences.getFloat("sensor_height", 45.0);
    tankDiameter = preferences.getFloat("diameter", 228.0);
    numTanks = preferences.getInt("tanks", 1);
    telemetryInterval = preferences.getInt("interval", 15);
    gsmNumbersList = preferences.getString("gsm_nums", "");

    // Hardware Pins loaded from NVS or default macros
    pinLed = preferences.getInt("pin_led", HARDWARE_DEFAULT_LED_PIN);
    ledActiveLow = preferences.getBool("led_act_low", HARDWARE_DEFAULT_LED_ACTIVE_LOW);
    pinTrig = preferences.getInt("pin_trig", HARDWARE_DEFAULT_TRIG_PIN);
    pinEcho = preferences.getInt("pin_echo", HARDWARE_DEFAULT_ECHO_PIN);
    preferences.end();

    // 3. Initialize Hardware Pins
    pinMode(pinTrig, OUTPUT);
    digitalWrite(pinTrig, LOW);
    pinMode(pinEcho, INPUT);
    pinMode(pinLed, OUTPUT);

    // STRICT RULE: Built-in LED is strictly and unconditionally OFF on power-up
    setLedHardware(false);

    Serial.printf("[HW] LED Pin: GPIO %d (Polarity: %s) -> Initially OFF\n", 
                  pinLed, ledActiveLow ? "Active-LOW" : "Active-HIGH");
    Serial.printf("[HW] Sensor TRIG: GPIO %d | ECHO: GPIO %d\n", pinTrig, pinEcho);
    Serial.printf("[SYS] Device ID: %s | Telemetry Interval: %d sec\n", deviceId.c_str(), telemetryInterval);
    if (lastOtaUrl.length() > 0) {
        Serial.printf("[SYS] Device Memory OTA Flag: Last Applied = %s%s\n", 
                      lastOtaUrl.c_str(), 
                      lastOtaVer.length() > 0 ? (" (v" + lastOtaVer + ")").c_str() : "");
    } else {
        Serial.println("[SYS] Device Memory OTA Flag: Initial/Factory Flash (No OTA stored yet).");
    }
    if (apiServerUrl.length() > 0) {
        Serial.printf("[SYS] Configured Web App URL: %s\n", apiServerUrl.c_str());
    } else {
        Serial.println("[SYS] Web App URL: Not set yet (will configure via Captive Portal).");
    }
    Serial.flush();

    // 4. Initialize GSM interface if enabled
    initGSM();



    // 6. Connect to Wi-Fi or launch AP portal
    if (wifiSSID.length() > 0) {
        Serial.printf("[SYS] Connecting to Wi-Fi SSID: '%s'...\n", wifiSSID.c_str());
        WiFi.mode(WIFI_STA);
        WiFi.setAutoReconnect(true);
        WiFi.persistent(true);
        WiFi.begin(wifiSSID.c_str(), wifiPASS.c_str());

        int attempts = 0;
        bool connected = false;
        while (attempts < 30) {
            if (WiFi.status() == WL_CONNECTED && WiFi.localIP() != IPAddress(0, 0, 0, 0)) {
                connected = true;
                break;
            }
            // LED strictly remains OFF during connection attempt (NO BLINKING)
            setLedHardware(false);
            delay(500);
            Serial.print(".");
            attempts++;
        }

        if (connected) {
            Serial.println("\n[SYS] Wi-Fi Successfully Connected!");
            Serial.print("[SYS] Assigned IP Address: ");
            Serial.println(WiFi.localIP());
            Serial.print("[SYS] Gateway: ");
            Serial.println(WiFi.gatewayIP());
            Serial.print("[SYS] DNS Server: ");
            Serial.println(WiFi.dnsIP());

            if (WiFi.dnsIP() == IPAddress(0, 0, 0, 0)) {
                Serial.println("[SYS] Configuring public fallback DNS (8.8.8.8, 1.1.1.1)...");
                WiFi.config(INADDR_NONE, INADDR_NONE, INADDR_NONE, IPAddress(8, 8, 8, 8), IPAddress(1, 1, 1, 1));
            }

            // Turn on built-in LED ONLY when actively connected to Wi-Fi (SOLID ON)
            setLedHardware(true);

            setupWebServer();
        } else {
            Serial.println("\n[SYS] Wi-Fi connection timed out. Entering AP Setup Portal...");
            setLedHardware(false);
            startConfigMode();
        }
    } else {
        Serial.println("[SYS] No Wi-Fi credentials found in NVS Flash. Entering AP Setup Portal...");
        setLedHardware(false);
        startConfigMode();
    }
}

// =========================================================================================
// MAIN EXECUTIVE LOOP (loop)
// =========================================================================================

void loop() {
    server.handleClient();

    if (isConfigMode) {
        // STRICT RULE: Built-in LED is strictly OFF in AP / Config mode
        setLedHardware(false);

        updateGSMModule();

        unsigned long currentMs = millis();
        if (currentMs - lastSensorMeasureTime > 3000) {
            lastSensorMeasureTime = currentMs;
            getFilteredDistance();
        }

        static unsigned long lastConfigWifiRetry = 0;
        if (wifiSSID.length() > 0 && currentMs - lastConfigWifiRetry > 30000) {
            lastConfigWifiRetry = currentMs;
            if (WiFi.status() == WL_CONNECTED && WiFi.localIP() != IPAddress(0, 0, 0, 0)) {
                isConfigMode = false;
                WiFi.softAPdisconnect(true);
                WiFi.mode(WIFI_STA);
                setLedHardware(true);
                Serial.println("\n[SYS] Wi-Fi restored! Automatically exited AP Mode.");
                Serial.print("[SYS] Assigned IP Address: ");
                Serial.println(WiFi.localIP());

                if (WiFi.dnsIP() == IPAddress(0, 0, 0, 0)) {
                    WiFi.config(INADDR_NONE, INADDR_NONE, INADDR_NONE, IPAddress(8, 8, 8, 8), IPAddress(1, 1, 1, 1));
                }

                setupWebServer();
            } else {
                Serial.printf("[AP Mode] Background scan: Retrying connection to '%s'...\n", wifiSSID.c_str());
                WiFi.begin(wifiSSID.c_str(), wifiPASS.c_str());
            }
        }
    } else {
        // Normal Station Mode Execution
        bool isWifiConnected = (WiFi.status() == WL_CONNECTED && WiFi.localIP() != IPAddress(0, 0, 0, 0));

        // CONTINUOUS LED ENFORCEMENT:
        // LED is solid ON ONLY when connected to Wi-Fi.
        // If Wi-Fi is lost, LED turns OFF immediately.
        setLedHardware(isWifiConnected);

        static bool wasConnected = true;

        if (!isWifiConnected) {
            unsigned long currentMs = millis();

            if (wasConnected) {
                wasConnected = false;
                Serial.println("[WIFI] Connection lost! LED switched OFF.");
            }

            if (firstWifiDisconnectTime == 0) {
                firstWifiDisconnectTime = currentMs ? currentMs : 1;
                Serial.println("[WIFI] Watchdog armed (5-min fallback to AP portal).");
            }

            // Clean reconnection sequence instead of raw WiFi.reconnect()
            if (currentMs - lastWifiReconnectAttempt > 8000) {
                lastWifiReconnectAttempt = currentMs;
                Serial.printf("[WIFI] Reconnecting to '%s'...\n", wifiSSID.c_str());
                WiFi.disconnect();
                delay(100);
                WiFi.begin(wifiSSID.c_str(), wifiPASS.c_str());
            }

            if (currentMs - firstWifiDisconnectTime > 300000) {
                Serial.println("[WIFI] Disconnected for 5 minutes. Launching AP Portal.");
                startConfigMode();
            }
        } else {
            if (!wasConnected) {
                wasConnected = true;
                firstWifiDisconnectTime = 0;
                Serial.println("\n[WIFI] Connection successfully restored!");
                Serial.print("[WIFI] IP Address: ");
                Serial.println(WiFi.localIP());
                Serial.print("[WIFI] Gateway: ");
                Serial.println(WiFi.gatewayIP());
                Serial.print("[WIFI] DNS Server: ");
                Serial.println(WiFi.dnsIP());

                if (WiFi.dnsIP() == IPAddress(0, 0, 0, 0)) {
                    Serial.println("[WIFI] Setting public fallback DNS (8.8.8.8, 1.1.1.1)...");
                    WiFi.config(INADDR_NONE, INADDR_NONE, INADDR_NONE, IPAddress(8, 8, 8, 8), IPAddress(1, 1, 1, 1));
                }

                setLedHardware(true);
                delay(300); // Settle network stack
            }
        }

        updateGSMModule();

        unsigned long currentMs = millis();

        // High-speed ultrasonic sampling & EMA filtering (every 150ms)
        if (currentMs - lastSensorMeasureTime > 150) {
            lastSensorMeasureTime = currentMs;
            getFilteredDistance();
        }

        // Periodic Telemetry Transmission (Interval: default 15s)
        if (currentMs - lastMeasureTime > ((unsigned long)telemetryInterval * 1000)) {
            lastMeasureTime = currentMs;

            float distance = (smoothedDistance >= 0.0) ? smoothedDistance : getFilteredDistance();

            // Core Water Level Calculations
            float depth = (tankHeight + sensorHeight) - distance;
            if (depth < 0.0) depth = 0.0;
            if (depth > tankHeight) depth = tankHeight;

            float radius = tankDiameter / 2.0;
            float volume = ((3.14159265 * radius * radius * depth) / 1000.0) * numTanks;

            currentDepthValue = depth;
            currentVolumeValue = volume;

            Serial.printf("[SENS] Distance: %.1f cm | Depth: %.1f cm | Volume: %.1f L\n",
                          distance, depth, volume);

            // Primary 24/7 HTTP REST Telemetry POST (Single-channel direct ingestion)
            sendHttpTelemetry(depth, volume);
        }
    }

    // Deferred Reboot Execution
    if (pendingReboot && millis() - rebootTimer > 2000) {
        Serial.println("[SYS] Rebooting now...");
        ESP.restart();
    }
}
