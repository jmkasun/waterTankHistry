/*
 * =========================================================================
 *                    IoT HydroSync - Water Level Controller
 *          Designed for ESP32 & ESP32-C3 series with HC-SR04/US-100
 * =========================================================================
 * 
 * Hardware Connections:
 * - HC-SR04 / US-100 TRIG Pin: GPIO 2 (Configurable)
 * - HC-SR04 / US-100 ECHO Pin: GPIO 3 (Configurable)
 * - VCC: 5V (or 3.3V for US-100 low voltage operations)
 * - GND: GND
 * 
 * Features:
 * - Persistent configurations (SSID, Password, Tank Height, Tank Diameter, Topic Prefix) in NVS.
 * - Wi-Fi Captive AP Configuration Portal (runs automatically when Wi-Fi is disconnected).
 * - Live Over-The-Air (OTA) configuration updates via MQTT.
 * - Exponential moving average (EMA) noise filter to prevent ultrasonic reading spikes.
 * - Graceful visual feedback using built-in LED blinking.
 */

    }

// --- Configures local AP for setting Wi-Fi ---
void startConfigMode() {
    isConfigMode = true;
    Serial.println("[SYS] Booting Config AP Portal...");
    if (wifiSSID != "") {
        // Run in dual AP + STA mode so we can auto-reconnect in the background
        WiFi.mode(WIFI_AP_STA);
        WiFi.begin(wifiSSID.c_str(), wifiPASS.c_str());
        Serial.println("[SYS] Configured as AP_STA. Still trying to connect to saved Wi-Fi in background...");
    } else {
        WiFi.mode(WIFI_AP);
    }
    WiFi.softAP("ESP32_HydroSync_AP");
    setupWebServer();
    Serial.println("[SYS] Captive Portal up! Connect to SSID: 'ESP32_HydroSync_AP' (Open Network)");
}

// --- High Precision Distance Measurement with Median Pre-Filter ---
float getDistanceRaw() {
    digitalWrite(TRIG_PIN, LOW);
    delayMicroseconds(4);
    digitalWrite(TRIG_PIN, HIGH);
    delayMicroseconds(10);
    digitalWrite(TRIG_PIN, LOW);
    
    // Echo pulse measurement (30ms timeout equivalent to ~5 meters range)
    long duration = pulseIn(ECHO_PIN, HIGH, 30000);
    if (duration == 0) {
        return -1.0; // Fail flag indicating sensor bounds exception
    }
    
    // Speed of sound calculation (0.0343 cm per microsecond / 2.0 for bounce trip)
    return (duration * 0.0343) / 2.0;
}

// --- Reliable Filtered Measurement Loop ---
float getFilteredDistance() {
    float raw = getDistanceRaw();
    if (raw < 0.0 || raw > (tankHeight + sensorHeight) + 20.0) {
        // Return current filtered spacing if sensor reads a failure glitch spike
        return (smoothedDistance >= 0.0) ? smoothedDistance : (tankHeight + sensorHeight);
    }
    
    if (smoothedDistance < 0.0) {
        // Warm up filter on first successful read
        smoothedDistance = raw;
    } else {
        // Apply EMA filter
        smoothedDistance = (raw * filterAlpha) + (smoothedDistance * (1.0 - filterAlpha));
    }
    return smoothedDistance;
}

// --- Remote HTTP OTA Firmware Flasher ---
void triggerHttpOta(String url) {
    Serial.println("[OTA] Initializing remote Over-The-Air firmware download...");
    Serial.println("[OTA] Firmware binary URL: " + url);
    
    // Wire up progress callback
    httpUpdate.onProgress([](int current, int total) {
        Serial.printf("[OTA] Downloading: %d%%\r", (current * 100) / total);
    });

    t_httpUpdate_return result;

    if (url.startsWith("https://")) {
        WiFiClientSecure clientSecure;
        clientSecure.setInsecure(); // Skip certificate validation for secure links (useful for GitHub Raw URLs)
        result = httpUpdate.update(clientSecure, url);
    } else {
        WiFiClient client;
        result = httpUpdate.update(client, url);
    }

    if (result == HTTP_UPDATE_FAILED) {
        Serial.printf("[OTA] Update failed! Error code: %d | Info: %s\n", 
                      httpUpdate.getLastError(), 
                      httpUpdate.getLastErrorString().c_str());
    } else if (result == HTTP_UPDATE_NO_UPDATES) {
        Serial.println("[OTA] No firmware differences detected by local host.");
    } else if (result == HTTP_UPDATE_OK) {
        Serial.println("[OTA] Firmware written successfully. Rebooting ESP32 device now...");
        delay(1500);
        ESP.restart();
    }
}

// --- MQTT Incoming Payload Processing Handler ---
void mqttCallback(char* topic, byte* payload, unsigned int length) {
    String topicStr = String(topic);
    String payloadStr = "";
    for (unsigned int i = 0; i < length; i++) {
        payloadStr += (char)payload[i];
    }
    payloadStr.trim();

    Serial.printf("[MQTT] Rx on: %s | Msg: %s\n", topic, payloadStr.c_str());

    preferences.begin(PREF_NAMESPACE, false);

    if (topicStr.endsWith("/config/height")) {
        float val = payloadStr.toFloat();
        if (val > 0) {
            tankHeight = val;
            preferences.putFloat("height", val);
            Serial.printf("[NVS] Tank height saved: %.1f cm\n", val);
        }
    } 
    else if (topicStr.endsWith("/config/sensor_height")) {
        float val = payloadStr.toFloat();
        if (val >= 0) {
            sensorHeight = val;
            preferences.putFloat("sensor_height", val);
            Serial.printf("[NVS] Sensor height offset saved: %.1f cm\n", val);
        }
    }
    else if (topicStr.endsWith("/config/diameter")) {
        float val = payloadStr.toFloat();
        if (val > 0) {
            tankDiameter = val;
            preferences.putFloat("diameter", val);
            Serial.printf("[NVS] Tank diameter saved: %.1f cm\n", val);
        }
    } 
    else if (topicStr.endsWith("/config/tanks")) {
        int val = payloadStr.toInt();
        if (val > 0) {
            numTanks = val;
            preferences.putInt("tanks", val);
            Serial.printf("[NVS] Number of tanks saved: %d\n", val);
        }
    }
    else if (topicStr.endsWith("/config/interval")) {
        int val = payloadStr.toInt();
        if (val >= 5) {
            telemetryInterval = val;
            preferences.putInt("interval", val);
            Serial.printf("[NVS] Telemetry interval saved: %d seconds\n", val);
        }
    }
    else if (topicStr.endsWith("/config/gsm_numbers")) {
        gsmNumbersList = payloadStr;
        preferences.putString("gsm_nums", payloadStr);
        Serial.printf("[NVS] Whitelisted GSM phone numbers list saved: %s\n", payloadStr.c_str());
    }
    else if (topicStr.endsWith("/config/api_url")) {
        apiServerUrl = payloadStr;
        preferences.putString("api_url", payloadStr);
        Serial.printf("[NVS] Web App Server HTTP URL saved: %s\n", payloadStr.c_str());
    }
    else if (topicStr.endsWith("/config/motor1_rate")) {
        float val = payloadStr.toFloat();
        if (val >= 0) {
            motor1Rate = val;
            preferences.putFloat("motor1_rate", val);
            Serial.printf("[NVS] Motor 1 Pump Rate saved: %.1f L/hr\n", val);
        }
    }
    else if (topicStr.endsWith("/config/motor2_rate")) {
        float val = payloadStr.toFloat();
        if (val >= 0) {
            motor2Rate = val;
            preferences.putFloat("motor2_rate", val);
            Serial.printf("[NVS] Motor 2 Pump Rate saved: %.1f L/hr\n", val);
        }
    }
    else if (topicStr.endsWith("/config/pump_threshold")) {
        float val = payloadStr.toFloat();
        if (val >= 0) {
            pumpThreshold = val;
            preferences.putFloat("pump_thresh", val);
            Serial.printf("[NVS] Pump Detection Threshold saved: %.1f L/hr\n", val);
        }
    }
    else if (topicStr.endsWith("/config/alert_min")) {
        float val = payloadStr.toFloat();
        if (val >= 0 && val <= 100) {
            alertMinLevel = val;
            preferences.putFloat("alert_min", val);
            Serial.printf("[NVS] Alert Min Level saved: %.1f %%\n", val);
        }
    }
    else if (topicStr.endsWith("/config/alert_max")) {
        float val = payloadStr.toFloat();
        if (val >= 0 && val <= 100) {
            alertMaxLevel = val;
            preferences.putFloat("alert_max", val);
            Serial.printf("[NVS] Alert Max Level saved: %.1f %%\n", val);
        }
    }
    else if (topicStr.endsWith("/cmd/ota")) {
        Serial.println("[OTA] OTA Trigger command received via MQTT!");
        if (payloadStr.startsWith("http://") || payloadStr.startsWith("https://")) {
            triggerHttpOta(payloadStr);
        } else {
            Serial.println("[OTA] Error: Firmware URL must start with http:// or https://");
        }
    }

    preferences.end();
}

        mqttClient.publish(topicMotor1.c_str(), motor1Str.c_str(), true);
        mqttClient.publish(topicMotor2.c_str(), motor2Str.c_str(), true);
        mqttClient.publish(topicThresh.c_str(), threshStr.c_str(), true);
        mqttClient.publish(topicAlertMin.c_str(), alertMinStr.c_str(), true);
        mqttClient.publish(topicAlertMax.c_str(), alertMaxStr.c_str(), true);
        mqttClient.publish(topicIp.c_str(), ipStr.c_str(), true);
        mqttClient.publish(topicVersion.c_str(), FIRMWARE_VERSION, true);

        // Accumulate configuration packet sizes with network overhead (~45 bytes per packet)
        totalBytesSent += (topicHeight.length() + heightStr.length() + 45);
        totalBytesSent += (topicSensorHeight.length() + sensorHeightStr.length() + 45);
        totalBytesSent += (topicDiameter.length() + diameterStr.length() + 45);
        totalBytesSent += (topicTanks.length() + tanksStr.length() + 45);
        totalBytesSent += (topicGsm.length() + gsmNumbersList.length() + 45);
        totalBytesSent += (topicInterval.length() + intervalStr.length() + 45);
        totalBytesSent += (topicApiUrl.length() + apiServerUrl.length() + 45);
        totalBytesSent += (topicMotor1.length() + motor1Str.length() + 45);
        totalBytesSent += (topicMotor2.length() + motor2Str.length() + 45);
        totalBytesSent += (topicThresh.length() + threshStr.length() + 45);
        totalBytesSent += (topicAlertMin.length() + alertMinStr.length() + 45);
        totalBytesSent += (topicAlertMax.length() + alertMaxStr.length() + 45);
        totalBytesSent += (topicIp.length() + ipStr.length() + 45);
        totalBytesSent += (topicVersion.length() + strlen(FIRMWARE_VERSION) + 45);

        Serial.println("[MQTT] Published current specs, phone whitelist, telemetry interval, API URL, consumption parameters, and local IP to broker.");
    }
}

unsigned long lastMqttReconnectAttempt = 0;

// --- MQTT Reconnect (Non-Blocking) ---
void connectMQTTNonBlocking() {
    unsigned long now = millis();
    if (now - lastMqttReconnectAttempt > 5000) {
        lastMqttReconnectAttempt = now;
        Serial.print("[MQTT] Attempting non-blocking connection to HiveMQ Broker...");
        String clientId = "HydroSyncClient-" + String(random(0xffff), HEX);
        if (mqttClient.connect(clientId.c_str())) {
            Serial.println("CONNECTED!");
            subscribeToConfig();
            publishCurrentConfig();
        } else {
            Serial.printf("FAILED (rc = %d). Will retry in 5 seconds without blocking physical controller.\n", mqttClient.state());
        }
    }
}

// --- GSM Whitelist & Call Handling Functions ---
bool isNumberAuthorized(String callingNumber) {
    if (callingNumber.length() == 0) return false;
    if (gsmNumbersList.length() == 0) return false;

    callingNumber.trim();
    String tempConfig = gsmNumbersList;
    tempConfig.trim();
    if (tempConfig.length() == 0) return false;

    int startIndex = 0;
    while (true) {
        int commaIndex = tempConfig.indexOf(',', startIndex);
        String savedNum = "";
        if (commaIndex == -1) {
            savedNum = tempConfig.substring(startIndex);
        } else {
            savedNum = tempConfig.substring(startIndex, commaIndex);
        }
        savedNum.trim();

        if (savedNum.length() > 0) {
            if (callingNumber == savedNum) {
                return true;
            }
            if (savedNum.length() >= 9 && callingNumber.length() >= 9) {
                String subSaved = savedNum.substring(savedNum.length() - 9);
                String subCalling = callingNumber.substring(callingNumber.length() - 9);
                if (subSaved == subCalling) {
                    return true;
                }
            }
        }

        if (commaIndex == -1) break;
        startIndex = commaIndex + 1;
    }
    return false;
}

void initGSM() {
    Serial.println("[GSM] Initialising SIM/GSM Serial Interface (9600 bps)...");
    SerialGSM.begin(9600, SERIAL_8N1, GSM_RX_PIN, GSM_TX_PIN);
    delay(1000);

    SerialGSM.println("AT");
    delay(200);
    SerialGSM.println("ATE0"); // Disable Echo
    delay(200);
    SerialGSM.println("AT+CLIP=1"); // Caller ID Presentation
    delay(200);
    SerialGSM.println("AT+CMGF=1"); // SMS Text mode
    delay(200);

    while (SerialGSM.available()) {
        SerialGSM.read(); // Clear input buffers
    }

    gsmInitialized = true;
    Serial.println("[GSM] Module successfully configured and ready for calls.");
}

void sendSMS(String number, String message) {
    Serial.printf("[GSM] Preparing to send SMS to: %s\n", number.c_str());
    SerialGSM.print("AT+CMGS=\"");
    SerialGSM.print(number);
    SerialGSM.println("\"");
    delay(500); // Allow modem processing time for '>' prompt
    
    SerialGSM.print(message);
    delay(200);
    SerialGSM.write(26); // Ctrl+Z submission flag character
    delay(500);
    Serial.println("[GSM] SMS submission sequence transmitted.");
}

void updateGSMModule() {
    if (!gsmInitialized) return;

    while (SerialGSM.available()) {
        char c = SerialGSM.read();
        if (c == '\n') {
            gsmBuffer.trim();
            if (gsmBuffer.length() > 0) {
                Serial.print("[GSM Rx] ");
                Serial.println(gsmBuffer);

                if (gsmBuffer.startsWith("+CLIP:")) {
                    Serial.println("[GSM] Incoming call detected. Parsing details...");
                    
                    int firstQuote = gsmBuffer.indexOf('"');
                    int secondQuote = gsmBuffer.indexOf('"', firstQuote + 1);

                    if (firstQuote != -1 && secondQuote != -1) {
                        String incomingNumber = gsmBuffer.substring(firstQuote + 1, secondQuote);
                        incomingNumber.trim();
                        Serial.printf("[GSM] Calling Number parsed: %s\n", incomingNumber.c_str());

                        // Quick hangup to keep it as a missed call & avoid call charges
                        SerialGSM.println("ATH");
                        Serial.println("[GSM] Hang up command sent (ATH).");

                        if (isNumberAuthorized(incomingNumber)) {
                            Serial.println("[GSM] Number Authorized! Sending status report SMS...");

                            float percentage = (currentDepthValue / tankHeight) * 100.0;
                            if (percentage < 0.0) percentage = 0.0;
                            if (percentage > 100.0) percentage = 100.0;

                            String sms = "HydroSync Water Level Report:\n";
                            sms += "- Depth: " + String(currentDepthValue, 1) + " cm\n";
                            sms += "- Sensor Margins: " + String((smoothedDistance >= 0.0) ? smoothedDistance : 0.0, 1) + " cm\n";
                            sms += "- Fill Rate: " + String(percentage, 0) + "%\n";
                            sms += "- Volume: " + String(currentVolumeValue, 0) + " Liters\n";
                            sms += "- Parallel Tanks: " + String(numTanks);

                            sendSMS(incomingNumber, sms);
                        } else {
                            Serial.println("[GSM] Number NOT authorized. Call ignored.");
                        }
                    }
                }
            }
            gsmBuffer = "";
        } else if (c != '\r') {
            gsmBuffer += c;
        }
    }
}

// --- Firmware Entry Configuration ---
void setup() {
    Serial.begin(115200);
    delay(500);
    Serial.println("\n====================================");
    Serial.println("   IoT HydroSync ESP32 Controller  ");
    Serial.println("====================================");

    pinMode(TRIG_PIN, OUTPUT);
    pinMode(ECHO_PIN, INPUT);
    pinMode(LED_PIN, OUTPUT);

    // Initialise Global GSM modem interface
    initGSM();

    // Load configurations from NVS Storage
    preferences.begin(PREF_NAMESPACE, true);
    bool isConfigured = preferences.getBool("configured", false);
    if (isConfigured) {
        wifiSSID = preferences.getString("ssid", "");
        wifiPASS = preferences.getString("pass", "");
    } else {
        wifiSSID = "Xiaomi_FC28";
        wifiPASS = "19990814";
    }
    tankHeight = preferences.getFloat("height", 200.0);
    sensorHeight = preferences.getFloat("sensor_height", 45.0);
    tankDiameter = preferences.getFloat("diameter", 228.0);
    numTanks = preferences.getInt("tanks", 1);
    telemetryInterval = preferences.getInt("interval", 15);
    mqttPrefix = preferences.getString("prefix", "mytank123");
    gsmNumbersList = preferences.getString("gsm_nums", "");
    apiServerUrl = preferences.getString("api_url", "");
    motor1Rate = preferences.getFloat("motor1_rate", 1000.0);
    motor2Rate = preferences.getFloat("motor2_rate", 5000.0);
    pumpThreshold = preferences.getFloat("pump_thresh", 2500.0);
    alertMinLevel = preferences.getFloat("alert_min", 20.0);
    alertMaxLevel = preferences.getFloat("alert_max", 90.0);
    preferences.end();

    Serial.printf("[NVS] Loaded WiFi SSID: '%s' | Password: '%s' | Configured: %s\n", 
                  wifiSSID.c_str(), wifiPASS.c_str(), isConfigured ? "YES" : "NO");
    Serial.printf("[NVS] Loaded Web App Server URL: '%s'\n", apiServerUrl.c_str());
    Serial.printf("[NVS] Tank specs: Height=%.1fcm | SensorOffset=%.1fcm | Diameter=%.1fcm | Tanks=%d | Interval=%d s\n", tankHeight, sensorHeight, tankDiameter, numTanks, telemetryInterval);
    Serial.printf("[NVS] Consumption: Motor1Rate=%.1f L/h | Motor2Rate=%.1f L/h | PumpThreshold=%.1f L/h\n", motor1Rate, motor2Rate, pumpThreshold);
    Serial.printf("[NVS] Alerts: Min=%.1f%% | Max=%.1f%%\n", alertMinLevel, alertMaxLevel);
    Serial.printf("[NVS] Topic config path prefix: %s\n", mqttPrefix.c_str());

    if (wifiSSID == "") {
        Serial.println("[SYS] No network credentials specified.");
        startConfigMode();
    } else {
        Serial.printf("[SYS] Configuring station mode. Connecting to Wi-Fi SSID: %s ...\n", wifiSSID.c_str());
        
        // Robust Wi-Fi STA Initialization sequence
        WiFi.persistent(false);
        WiFi.disconnect();
        delay(200);
        WiFi.mode(WIFI_STA);
        delay(200);
        
        WiFi.begin(wifiSSID.c_str(), wifiPASS.c_str());
        
        int attempts = 0;
        bool connected = false;
        // Increase loop attempts to 60 (60 * 500ms = 30-second window for DHCP)
        while (attempts < 60) {
            if (WiFi.status() == WL_CONNECTED) {
                connected = true;
                break;
            }
            // Flash LED quickly to represent connection loop
            digitalWrite(LED_PIN, !digitalRead(LED_PIN));
            delay(500);
            Serial.print(".");
            attempts++;
        }
        
        if (connected) {
            digitalWrite(LED_PIN, HIGH); // Turn on steady when connected
            Serial.println("\n[SYS] Network Connected OK!");
            Serial.print("[SYS] IP Address: ");
            Serial.println(WiFi.localIP());
            
            setupWebServer(); // Initialize Local WebServer (Config Portal)
            
            mqttClient.setServer(mqttServer, mqttPort);
            mqttClient.setCallback(mqttCallback);
            connectMQTTNonBlocking();
        } else {
            Serial.println("\n[SYS] Network connection timeout. Entering AP Setup Web Portal...");
            startConfigMode();
        }
    }
}

// --- Continuous System Application Loop ---
void loop() {
    // Process local WebServer requests (Configs) in both modes
    server.handleClient();

    if (isConfigMode) {
        // Fast blinking sequence representing local Config mode active (Double Flash)
        unsigned long currentMs = millis();
        if (currentMs - lastLedBlinkTime > 150) {
            lastLedBlinkTime = currentMs;
            ledState = !ledState;
            digitalWrite(LED_PIN, ledState);
        }

        // Maintain and process incoming SIM/GSM serial communications (so SMS queries still work!)
        updateGSMModule();

        // Feed the filter continuously in the background
        if (currentMs - lastSensorMeasureTime > 150) {
            lastSensorMeasureTime = currentMs;
            getFilteredDistance();
        }

        // Update GSM cache values occasionally
        if (currentMs - lastMeasureTime > 30000) {
            lastMeasureTime = currentMs;
            float distance = (smoothedDistance >= 0.0) ? smoothedDistance : getFilteredDistance();
            float depth = (tankHeight + sensorHeight) - distance;
            if (depth < 0.0) depth = 0.0;
            if (depth > tankHeight) depth = tankHeight;

            float radius = tankDiameter / 2.0;
            float volume = ((3.14159265 * radius * radius * depth) / 1000.0) * numTanks;

            currentDepthValue = depth;
            currentVolumeValue = volume;
            Serial.printf("[SENS Config AP] Distance: %.1f cm | Depth: %.1f cm | Volume: %.1f L\n", distance, depth, volume);
        }

        // Check if Wi-Fi successfully connected in the background
        if (wifiSSID != "" && WiFi.status() == WL_CONNECTED) {
            Serial.println("\n[SYS] Background Wi-Fi connected successfully!");
            Serial.print("[SYS] IP Address: ");
            Serial.println(WiFi.localIP());

            // Close the AP mode to save power and clear the airwaves, switch to pure STA station mode
            WiFi.softAPdisconnect(true);
            WiFi.mode(WIFI_STA);

            // Configure MQTT
            mqttClient.setServer(mqttServer, mqttPort);
            mqttClient.setCallback(mqttCallback);
            connectMQTTNonBlocking();

            // Set LED stable high (connected)
            digitalWrite(LED_PIN, HIGH);

            // Exit Config mode
            isConfigMode = false;
            Serial.println("[SYS] Transitioned from Config AP Mode to Normal Station Mode automatically.");
        }
    } else {
        // Keep MQTT live without blocking
        if (!mqttClient.connected()) {
            connectMQTTNonBlocking();
        } else {
            mqttClient.loop();
        }

        // Maintain and process incoming SIM/GSM serial communications
        updateGSMModule();

        unsigned long currentMs = millis();

        // Handle deferred reboot safely after some buffer delay
        if (pendingReboot && (currentMs - rebootTimer > 5000)) {
            Serial.println("[SYS] Applying fresh WiFi configuration. Rebooting device now...");
            delay(1000);
            ESP.restart();
        }

        // Feed the filter continuously in the background every 150ms for high responsiveness!
        if (currentMs - lastSensorMeasureTime > 150) {
            lastSensorMeasureTime = currentMs;
            getFilteredDistance(); // This continuously reads raw distance and updates smoothedDistance
        }

        // Perform packet transmission based on configured telemetry interval (default 15s)
        if (currentMs - lastMeasureTime > ((unsigned long)telemetryInterval * 1000)) {
            lastMeasureTime = currentMs;

            // Use the continuously updated smoothedDistance (or compute one if not ready)
            float distance = (smoothedDistance >= 0.0) ? smoothedDistance : getFilteredDistance();
            
            // Core calculation formulas: Depth (cm) and Volume (Liters)
            float depth = (tankHeight + sensorHeight) - distance;
            if (depth < 0.0) depth = 0.0;
            if (depth > tankHeight) depth = tankHeight;

            float radius = tankDiameter / 2.0;
            float volume = ((3.14159265 * radius * radius * depth) / 1000.0) * numTanks;

            // Cache current values securely for GSM SMS queries
            currentDepthValue = depth;
            currentVolumeValue = volume;

            Serial.printf("[SENS] Distance: %.1f cm | Calculated Depth: %.1f cm | Calculated Volume (All Tanks): %.1f L\n", 
                          distance, depth, volume);

            if (mqttClient.connected()) {
                // Publish metrics to HydroSync specified channels
                String topicDepth = mqttPrefix + "/status/level";
                String topicVolume = mqttPrefix + "/status/volume";
                String topicData = mqttPrefix + "/status/data_usage";

                String depthStr = String(depth, 1);
                String volumeStr = String(volume, 1);

                mqttClient.publish(topicDepth.c_str(), depthStr.c_str(), true);
                mqttClient.publish(topicVolume.c_str(), volumeStr.c_str(), true);

                // Increment data overhead (~45 bytes TCP/IP/MQTT envelope + topic length + value string length)
                totalBytesSent += (topicDepth.length() + depthStr.length() + 45);
                totalBytesSent += (topicVolume.length() + volumeStr.length() + 45);

                String dataStr = String(totalBytesSent);
                mqttClient.publish(topicData.c_str(), dataStr.c_str(), true);
                totalBytesSent += (topicData.length() + dataStr.length() + 45);

                Serial.printf("[MQTT] Telemetry parameters successfully transmitted to broker. Cumulative Data Usage: %lu bytes.\n", totalBytesSent);

                // Pulse the board LED briefly to acknowledge sensor packet transmission
                digitalWrite(LED_PIN, LOW);
                delay(100);
                digitalWrite(LED_PIN, HIGH);
            }

            // Send HTTP POST telemetry to keep server logging 24/7 (even when dashboard is closed)
            if (WiFi.status() == WL_CONNECTED && apiServerUrl.length() > 0) {
                HTTPClient http;
                String endpoint = apiServerUrl;
                if (!endpoint.endsWith("/")) {
                    endpoint += "/";
                }
                endpoint += "api/telemetry";
                
                Serial.println("[HTTP] Sending telemetry POST to: " + endpoint);
                
                String jsonPayload = "{\"device_id\":\"" + mqttPrefix + "\",\"level\":" + String(depth, 1) + ",\"volume\":" + String(volume, 1) + ",\"data_usage\":" + String(totalBytesSent) + ",\"version\":\"" + String(FIRMWARE_VERSION) + "\"}";
                int httpResponseCode = -1;

                if (endpoint.startsWith("https://")) {
                    WiFiClientSecure clientSecure;
                    clientSecure.setInsecure(); // Skip certificate verification for SSL
                    http.begin(clientSecure, endpoint);
                    http.addHeader("Content-Type", "application/json");
                    httpResponseCode = http.POST(jsonPayload);
                } else {
                    WiFiClient client;
                    http.begin(client, endpoint);
                    http.addHeader("Content-Type", "application/json");
                    httpResponseCode = http.POST(jsonPayload);
                }

                if (httpResponseCode > 0) {
                    Serial.printf("[HTTP] Telemetry POST success, response: %d\n", httpResponseCode);
                } else {
                    Serial.printf("[HTTP] Telemetry POST failed, error: %s\n", http.errorToString(httpResponseCode).c_str());
                }
                http.end();
            }
        }
    }
}
